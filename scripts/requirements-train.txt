# Training-only dependencies for the Lean 4 LoRA fine-tune (scripts/lean_lora_sft.py).

# universal lock across the 3.14 main venv and the 3.12 .venv-lean, and every

# Bring-up on the training box (Python 3.10-3.12, CUDA):
#   python -m venv .venv-train && . .venv-train/bin/activate
#   pip install -r scripts/requirements-train.txt
#   pip install -e .            # smolbench itself, for smolbench.deduction.lean.sft
#
# The eval/serving path never imports any of these (the LoRA is served through
# vLLM on the EC2 provider), so the offline test suite and both eval venvs are
# unaffected.
