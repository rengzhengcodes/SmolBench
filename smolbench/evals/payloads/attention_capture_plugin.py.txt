"""smolbench_attn: capture the answer tokens' attention rows inside vLLM's prefill.

A vLLM *general plugin* (entry point ``smolbench_attn = smolbench_attn:register``,
loaded in every process including tensor-parallel workers). It wraps
``Attention.forward`` so that, while a job is armed, the post-RoPE queries and
keys the layer receives are turned into the softmax rows of the job's
positions -- the very rows the kernel used, because causal attention at a
position depends only on the prefix and this is the same checkpoint in the
same serving stack that generated the sequence.

Contract: ``smolbench/evals/payloads/ATTENTION_CAPTURE_CONTRACT.md``. In short:
the box's agent writes ``$SMOLBENCH_ATTN_DIR/current.json`` (job id, seq,
n_tokens, positions, per-rank presigned PUT URLs) and then sends ONE
completions request whose prompt is the stored token ids with
``max_tokens=1``. The serve is ``--enforce-eager`` (this wrapper is plain
Python; it cannot live inside a fullgraph compile), ``--max-num-seqs 1`` and
``--max-num-batched-tokens <max_model_len>`` (one un-chunked prefill), so
``query.shape[0] == n_tokens`` identifies the job's forward.

Per layer the rows honour: the impl's ``scale``; the layer's sliding window
(vLLM/flash-attn convention ``window=(w-1, 0)``: keys ``j >= p-(w-1)``);
grouped-query head mapping; ``logits_soft_cap`` (tanh) when the impl has one;
attention sinks (gpt-oss: one extra softmax column per head, its mass stored
separately so the row over tokens is exactly the kernel's); KV sharing
(Gemma-4: ``key is None`` -- the target layer's keys/values, cached earlier in
the same forward, are used). Mamba blocks never call in; the captured layer
indices are recorded.

Multi-head latent attention (DeepSeek-V3.1): ``MLAAttention.forward`` is wrapped
as well, with the same job protocol and the same gate. That layer receives the
post-RoPE queries ``q [T, H, P+R]``, the latent ``kv_c_normed [T, L]`` and the
shared rotated ``k_pe [T, 1, R]``, so the per-head keys and values are rebuilt
the way vLLM's prefill path builds them: the layer's own ``kv_b_proj`` expands
the latent into ``[k_nope | v]`` per head and ``k_pe`` is broadcast over the
heads. The layer's ``num_kv_heads``/``head_size`` describe the latent cache, not
these keys, so an ``_MLAView`` carries the per-head shapes and the scale (read
from the layer, never recomputed) into the shared row, mass and gate code.
Every condition under which the rebuilt keys could differ from the kernel's
(decode tokens in the batch, chunked context, FP8 prefill, context parallelism,
sparse attention, a sliding window, a quantised output, an inconsistent scale or
unexpected shapes) fails the job through its done file; the reason starts with
``MLA guard <name>``. The meta then records ``attention`` ("mha", "mla" or
"mixed"); a job with MLA layers also records ``qk_head_dim``, ``v_head_dim``,
``kv_lora_rank`` and ``head_order`` (how the ranks' heads concatenate).

Self-consistency gate, every layer, every job: ``rows @ V`` at the captured
positions must match the kernel's own output rows from the same forward
(max relative difference <= ``SMOLBENCH_ATTN_GATE_TOL``, default 3e-2 for
bf16). A failed gate fails the job: nothing is uploaded, the done file says
why. The gate is what proves the wrapper read the right tensors with the
right masks, sinks and scale; without it a silently wrong capture would look
exactly like a right one.

Mode ``mass`` (the study's actual question -- do response tokens attend to
the whitespace padding less than uniform dispersion predicts?): the job also
carries a key CLASS per token and ``query_start``; for EVERY completion query
the row is reduced on the box to its mass per class (4 classes), its entropy
(over tokens plus the sink column), the sink mass and its prompt entropy (the
attention restricted to the prompt keys, renormalised to 1; the sink is not a
prompt key; NaN where no prompt key is attendable), in query chunks that
keep the float32 logits under ~1 GB. The gate rows at ``positions`` validate
the same arithmetic. No T x T matrix ever leaves the GPU.

Never raises into vLLM: every failure is recorded in the job's done file and
serving continues.
"""
import io
import json
import math
import os
import re
import threading
import time
import types

_STATE = {
    "patched": False,
    "job": None,          # the armed job dict (from current.json), or None
    "job_mtime": None,    # mtime of current.json when `job` was read
    "session": None,      # per-forward capture in progress
    "done_seq": -1,       # seq of the last job finalised (ok or failed) on this rank
    "lock": threading.Lock(),
}

_LAYER_IDX = re.compile(r"\.layers\.(\d+)\.")


def _attn_dir():
    return os.environ.get("SMOLBENCH_ATTN_DIR", "/root/.cache/huggingface/attn")


def _gate_tol():
    return float(os.environ.get("SMOLBENCH_ATTN_GATE_TOL", "3e-2"))


def _rank():
    try:
        from vllm.distributed.parallel_state import get_tensor_model_parallel_rank

        return int(get_tensor_model_parallel_rank())
    except Exception:  # noqa: BLE001 -- single-process fallback (tests, tp=1 before init)
        return 0


def _log(msg):
    print("[smolbench_attn r%d] %s" % (_rank(), msg), flush=True)


# --------------------------------------------------------------------------- jobs
def _read_job():
    """Return the armed job (re-read when current.json changed), else None."""
    path = os.path.join(_attn_dir(), "current.json")
    try:
        mtime = os.stat(path).st_mtime_ns
    except OSError:
        _STATE["job"] = None
        _STATE["job_mtime"] = None
        return None
    if mtime != _STATE["job_mtime"]:
        try:
            with open(path) as fh:
                job = json.load(fh)
            job["positions"] = [int(p) for p in job["positions"]]
            job["n_tokens"] = int(job["n_tokens"])
            job["seq"] = int(job["seq"])
            job["mode"] = str(job.get("mode") or "rows")
            if job["mode"] == "mass":
                job["query_start"] = int(job["query_start"])
                job["classes"] = [int(c) for c in job["classes"]]
                if len(job["classes"]) != job["n_tokens"]:
                    raise ValueError("classes has %d entries for %d tokens"
                                     % (len(job["classes"]), job["n_tokens"]))
                if not (0 <= job["query_start"] < job["n_tokens"]):
                    raise ValueError("query_start %d outside sequence" % job["query_start"])
            elif job["mode"] != "rows":
                raise ValueError("unknown mode %r" % job["mode"])
            _STATE["job"] = job
        except Exception as exc:  # noqa: BLE001 -- a half-written file is retried next call
            _log("current.json unreadable: %r" % (exc,))
            _STATE["job"] = None
        _STATE["job_mtime"] = mtime
    job = _STATE["job"]
    if job is not None and job["seq"] <= _STATE["done_seq"]:
        return None
    return job


def _write_done(job, ok, error=None, gate_max_rel=None, nbytes=0, uploaded=False, http=None):
    done_dir = os.path.join(_attn_dir(), "done")
    os.makedirs(done_dir, exist_ok=True)
    path = os.path.join(done_dir, "%s.r%d.json" % (job["job_id"], _rank()))
    body = {"ok": bool(ok), "error": error, "gate_max_rel": gate_max_rel,
            "bytes": int(nbytes), "uploaded": bool(uploaded), "http_status": http,
            "seq": job["seq"], "rank": _rank()}
    tmp = path + ".tmp"
    with open(tmp, "w") as fh:
        json.dump(body, fh)
    os.replace(tmp, path)
    _STATE["done_seq"] = max(_STATE["done_seq"], job["seq"])
    _STATE["session"] = None


# --------------------------------------------------------------------------- rows
def _layer_index(name):
    m = _LAYER_IDX.search(name)
    return int(m.group(1)) if m else -1


def _attention_layers(classes):
    """All layers of the given classes in the static forward context, in model order.

    Parameters
    ----------
    classes : tuple of type
        The attention classes whose instances are captured (``Attention``, and
        ``MLAAttention`` when vLLM has it).

    Returns
    -------
    (list of str, dict)
        The layer names in ``no_compile_layers`` (dict) order, and the layers by name.
        The union over all classes: a job finalizes only after every one of them.
    """
    from vllm.forward_context import get_forward_context

    layers = get_forward_context().no_compile_layers
    names = [n for n, layer in layers.items() if isinstance(layer, classes)]
    return names, {n: layers[n] for n in names}


def _rows_for(layer, q3, k3, positions):
    """Softmax rows of `positions` over all keys, honouring window, GQA, softcap, sinks.

    Returns (rows float32 [P, H, T], sink float32 [P, H] or None).
    """
    import torch

    n = q3.shape[0]
    heads = layer.num_heads
    kv_heads = layer.num_kv_heads
    group = heads // kv_heads
    pos = torch.tensor(positions, device=q3.device, dtype=torch.long)
    qp = q3[pos].float()                                   # [P, H, D]
    qp = qp.view(len(positions), kv_heads, group, -1)       # [P, Hkv, g, D]
    kf = k3.float().permute(1, 0, 2)                        # [Hkv, T, D]
    scale = float(getattr(layer.impl, "scale", layer.head_size ** -0.5))
    logits = torch.einsum("pkgd,knd->pkgn", qp, kf) * scale  # [P, Hkv, g, T]
    logits = logits.reshape(len(positions), heads, n)       # [P, H, T]
    cap = float(getattr(layer.impl, "logits_soft_cap", 0) or 0)
    if cap > 0:
        logits = cap * torch.tanh(logits / cap)
    j = torch.arange(n, device=q3.device)
    mask = j[None, :] > pos[:, None]                        # future keys
    window = layer.sliding_window
    if window is not None and int(window) > 0:
        mask |= j[None, :] < (pos[:, None] - (int(window) - 1))
    logits = logits.masked_fill(mask[:, None, :], float("-inf"))
    sinks = getattr(layer.impl, "sinks", None) if getattr(layer, "has_sink", False) else None
    if sinks is not None:
        s = sinks.detach().float().view(1, heads, 1).expand(len(positions), heads, 1)
        full = torch.softmax(torch.cat([logits, s], dim=-1), dim=-1)
        return full[..., :n], full[..., n]
    return torch.softmax(logits, dim=-1), None


#: Float32 logits elements per query chunk (~1 GB); chunks keep P x T off the GPU.
MASS_CHUNK_ELEMENTS = 1 << 28
N_CLASSES = 4
#: Capture format version written to every meta json. Version 2 added the
#: mass-mode `prompt_entropy` array; a meta without the field is version 1.
CAPTURE_VERSION = 2
PUT_LANDED_EARLIER = "412-landed-earlier"  # 412 after an unclear try: it stored


def _logits_for(layer, q3, k3, pos):
    """Masked, scaled, soft-capped logits [P, H, T] for query positions `pos` (a tensor)."""
    import torch

    n = q3.shape[0]
    heads = layer.num_heads
    kv_heads = layer.num_kv_heads
    group = heads // kv_heads
    qp = q3[pos].float().view(len(pos), kv_heads, group, -1)
    kf = k3.float().permute(1, 0, 2)
    scale = float(getattr(layer.impl, "scale", layer.head_size ** -0.5))
    logits = torch.einsum("pkgd,knd->pkgn", qp, kf) * scale
    logits = logits.reshape(len(pos), heads, n)
    cap = float(getattr(layer.impl, "logits_soft_cap", 0) or 0)
    if cap > 0:
        logits = cap * torch.tanh(logits / cap)
    j = torch.arange(n, device=q3.device)
    mask = j[None, :] > pos[:, None]
    window = layer.sliding_window
    if window is not None and int(window) > 0:
        mask |= j[None, :] < (pos[:, None] - (int(window) - 1))
    return logits.masked_fill(mask[:, None, :], float("-inf"))


def _mass_for(layer, q3, k3, query_start, classes):
    """Per-class attention mass and row entropies for every query >= query_start.

    Returns (mass float16 [P, H, C], entropy float16 [P, H], sink float16 [P, H] or None,
    prompt_entropy float16 [P, H]). prompt_entropy is the entropy in nats of the
    row restricted to the prompt keys [0, query_start) and renormalised to 1
    (NaN where the row has no attendable prompt key).
    """
    import torch

    n = q3.shape[0]
    heads = layer.num_heads
    onehot = torch.zeros(n, N_CLASSES, device=q3.device, dtype=torch.float32)
    onehot[torch.arange(n, device=q3.device), classes] = 1.0
    sinks = getattr(layer.impl, "sinks", None) if getattr(layer, "has_sink", False) else None
    s = sinks.detach().float().view(1, heads, 1) if sinks is not None else None
    chunk = max(1, MASS_CHUNK_ELEMENTS // (heads * n))
    total = n - query_start
    mass = torch.empty(total, heads, N_CLASSES, dtype=torch.float16, device=q3.device)
    entropy = torch.empty(total, heads, dtype=torch.float16, device=q3.device)
    prompt_entropy = torch.empty(total, heads, dtype=torch.float16, device=q3.device)
    sink = torch.empty(total, heads, dtype=torch.float16, device=q3.device) if s is not None else None
    for start in range(query_start, n, chunk):
        pos = torch.arange(start, min(start + chunk, n), device=q3.device)
        logits = _logits_for(layer, q3, k3, pos)                       # [p, H, T]
        # Prompt entropy: a softmax over a key subset is the renormalised subset
        # of the full softmax, so use the subset's logits (log-softmax), taken
        # before the sink column joins: the sink is not a prompt key. Masked
        # keys are -inf and must add exactly 0 (0 * -inf would be nan). A row
        # whose subset logsumexp is not finite (empty subset, or every prompt
        # key masked by the window) has no defined value: NaN.
        pl = logits[..., :query_start]
        if query_start > 0:
            lse = torch.logsumexp(pl, dim=-1, keepdim=True)             # [p, H, 1]
            ok = torch.isfinite(lse)
            logp = pl - torch.where(ok, lse, torch.zeros_like(lse))
            pp = torch.exp(logp)
            pe = -torch.where(pp > 0, pp * logp, torch.zeros_like(pp)).sum(-1)
            pe = torch.where(ok[..., 0], pe, torch.full_like(pe, float("nan")))
        else:
            pe = torch.full(logits.shape[:2], float("nan"), device=q3.device)
        del pl
        if s is not None:
            logits = torch.cat([logits, s.expand(len(pos), heads, 1)], dim=-1)
        probs = torch.softmax(logits, dim=-1)
        ent = -(probs * torch.log(probs.clamp_min(1e-30))).sum(-1)      # [p, H]
        tok = probs[..., :n]
        m = torch.einsum("phn,nc->phc", tok, onehot)
        sl = slice(start - query_start, start - query_start + len(pos))
        mass[sl] = m.to(torch.float16)
        entropy[sl] = ent.to(torch.float16)
        prompt_entropy[sl] = pe.to(torch.float16)
        if sink is not None:
            sink[sl] = probs[..., n].to(torch.float16)
        del logits, probs, tok, m, pe
    return mass, entropy, sink, prompt_entropy


def _gate(layer, rows, v3, out_p):
    """Max abs/relative difference between rows @ V and the kernel's output rows."""
    import torch

    heads = layer.num_heads
    kv_heads = layer.num_kv_heads
    group = heads // kv_heads
    p = rows.shape[0]
    r = rows.view(p, kv_heads, group, -1)                    # [P, Hkv, g, T]
    vf = v3.float().permute(1, 0, 2)                        # [Hkv, T, Dv]
    o = torch.einsum("pkgn,knd->pkgd", r, vf).reshape(p, heads, -1)
    ref = out_p.float().reshape(p, heads, -1)
    diff = (o - ref).abs()
    max_abs = float(diff.max())
    denom = float(ref.abs().max()) + 1e-6
    return max_abs, max_abs / denom


# --------------------------------------------------------------------------- capture
def _capture_prelude(classes, layer, n):
    """Shared start of a capture: the armed job and this forward's session, or None.

    Runs for every wrapped layer on every forward, so it returns before touching
    anything else unless the forward is the job's own (``n == n_tokens``).

    Parameters
    ----------
    classes : tuple of type
        Attention classes of the model; their instances' count decides when the
        job is complete.
    layer : object
        The layer being called (only ``layer_name`` is read).
    n : int
        Number of tokens of this forward (the leading dimension of its queries).

    Returns
    -------
    (dict, dict) or None
        ``(job, session)``; None when nothing is armed, the length differs, the
        positions are invalid (the job is then failed), the session already
        failed, or this layer was already captured.
    """
    job = _read_job()
    if job is None:
        return None
    if n != job["n_tokens"]:
        return None
    if not job["positions"] or max(job["positions"]) >= n or min(job["positions"]) < 0:
        _write_done(job, False, error="positions %r outside sequence of %d tokens"
                    % (job["positions"], n))
        return None

    session = _STATE["session"]
    if session is None or session["seq"] != job["seq"]:
        names, layers = _attention_layers(classes)
        targets = {
            getattr(layers[nm], "kv_sharing_target_layer_name", None) for nm in names
        }
        targets.discard(None)
        session = {"seq": job["seq"], "job": job, "names": names, "targets": targets,
                   "kv": {}, "captured": {}, "gate": {}, "t0": time.time(),
                   "failed": None}
        _STATE["session"] = session
    if session["failed"] is not None:
        return None
    if layer.layer_name in session["captured"]:
        return None  # a second forward with the same token count; ignore
    return job, session


def _attention_adapter(layer, job, session, n, query, key, value, output):
    """Adapter for a plain ``Attention`` layer: its tensors in the shared tail's form.

    Returns
    -------
    tuple or None
        ``(view, q3, k3, v3, out2d, shared_from)``; the view is the layer itself.
        None after failing the job (a KV-sharing layer whose target was not cached).
    """
    name = layer.layer_name
    heads, head_size = layer.num_heads, layer.head_size
    head_size_v = getattr(layer, "head_size_v", head_size)
    q3 = query.view(n, heads, head_size)
    if key is None or value is None:
        target = getattr(layer, "kv_sharing_target_layer_name", None)
        if target is None or target not in session["kv"]:
            session["failed"] = "layer %s has no key/value and no cached KV target (%r)" % (name, target)
            _write_done(job, False, error=session["failed"])
            return None
        k3, v3 = session["kv"][target]
        shared_from = _layer_index(target)
    else:
        k3 = key.view(n, layer.num_kv_heads, head_size)
        v3 = value.view(n, layer.num_kv_heads, head_size_v)
        shared_from = -1
        if name in session["targets"]:
            session["kv"][name] = (k3, v3)
    return layer, q3, k3, v3, output.view(n, heads * head_size_v), shared_from


def _MLAView(layer):
    """Per-head view of an ``MLAAttention`` layer for the shared row/mass/gate code.

    The raw layer must never reach those helpers: its ``num_kv_heads`` (1) and
    ``head_size`` (``kv_lora_rank + qk_rope_head_dim``) describe the latent
    cache, not the per-head keys the kernel's logits were computed from. The
    view states the per-head shapes instead (every head has its own rebuilt key,
    so ``num_kv_heads == num_heads``), has no sink, window or soft cap, and
    carries the scale READ from the layer: the gate then proves it is the scale
    the kernel used, which a recomputed ``head_size ** -0.5`` could not.

    Parameters
    ----------
    layer : object
        A vLLM ``MLAAttention`` (this rank's shard: ``num_heads`` is local).

    Returns
    -------
    types.SimpleNamespace
        ``layer_name, num_heads, num_kv_heads, head_size`` (= ``P + R``),
        ``head_size_v``, ``sliding_window, has_sink, kv_sharing_target_layer_name,
        impl`` (``scale, logits_soft_cap, sinks``), ``attention`` ("mla"),
        ``qk_head_dim, v_head_dim, kv_lora_rank``.
    """
    p, r = int(layer.qk_nope_head_dim), int(layer.qk_rope_head_dim)
    return types.SimpleNamespace(
        layer_name=layer.layer_name,
        num_heads=int(layer.num_heads),
        num_kv_heads=int(layer.num_heads),
        head_size=p + r,
        head_size_v=int(layer.v_head_dim),
        sliding_window=None,
        has_sink=False,
        kv_sharing_target_layer_name=None,
        impl=types.SimpleNamespace(scale=float(layer.impl.scale), logits_soft_cap=0, sinks=None),
        attention="mla",
        qk_head_dim=int(layer.qk_head_dim),
        v_head_dim=int(layer.v_head_dim),
        kv_lora_rank=int(layer.kv_lora_rank),
    )


def _mla_kv(layer, kv_c_normed, k_pe, n):
    """Per-head keys and values of the first ``n`` tokens, as vLLM's prefill builds them.

    The layer's own ``kv_b_proj`` expands the latent into ``[k_nope | v]`` per
    head, and the shared rotated ``k_pe`` is broadcast over the heads. It is the
    same GEMM the kernel's keys came from, so the rebuilt keys match the
    kernel's; the absorbed ``W_UK_T``/``W_UV`` matrices belong to the
    decode path and are not used.

    Parameters
    ----------
    layer : object
        The ``MLAAttention`` layer (``kv_b_proj``, ``num_heads``, head dims).
    kv_c_normed : torch.Tensor
        ``[>= n, L]`` normalised latent.
    k_pe : torch.Tensor
        ``[>= n, 1, R]`` (or ``[>= n, R]``) rotated positional key.
    n : int
        Number of tokens to rebuild.

    Returns
    -------
    (torch.Tensor, torch.Tensor)
        ``k [n, H, P + R]`` and ``v [n, H, V]``.
    """
    import torch

    heads = int(layer.num_heads)
    p, r, v_dim = int(layer.qk_nope_head_dim), int(layer.qk_rope_head_dim), int(layer.v_head_dim)
    kv = layer.kv_b_proj(kv_c_normed[:n])[0].view(n, heads, p + v_dim)
    k_nope, v = kv.split([p, v_dim], dim=-1)
    k = torch.empty(n, heads, p + r, dtype=k_nope.dtype, device=k_nope.device)
    k[..., :p] = k_nope
    # reshape, not view: k_pe may be a non-contiguous slice; it is a view when it can be.
    k[..., p:] = k_pe[:n].reshape(n, 1, r)
    return k, v


def _mla_layer_metadata(name):
    """The per-layer attention metadata of the current forward, or None when absent.

    vLLM keeps it in the forward context as a dict by layer name, as a list whose
    first item is that dict (speculative decoding), or as one object.
    """
    from vllm.forward_context import get_forward_context

    md = get_forward_context().attn_metadata
    if isinstance(md, (list, tuple)):
        md = md[0] if md else None
    if isinstance(md, dict):
        return md.get(name)
    return md


def _mla_guard(layer, q, kv_c_normed, k_pe, output, n):
    """First failed MLA guard as ``(guard_name, detail)``, or None when every guard passes.

    Each guard names a condition under which the keys rebuilt from the latent
    could differ from the keys the kernel used while the self-consistency gate
    still passes or never runs; the order is fixed so a job fails with the
    most basic reason first.
    """
    import torch

    name = layer.layer_name
    md = _mla_layer_metadata(name)
    if md is None:
        return "prefill-only", "no attention metadata for this layer in the forward context"
    if getattr(md, "num_decode_tokens", None) != 0:
        return "prefill-only", "num_decode_tokens=%r, expected 0" % (getattr(md, "num_decode_tokens", None),)
    prefill = getattr(md, "prefill", None)
    if prefill is None:
        return "prefill-only", "metadata has no prefill part"
    if getattr(prefill, "chunked_context", None) is not None:
        return "chunked-context", "the prefill reads its context in chunks"
    q_dtype = getattr(prefill, "q_data_type", None)
    if layer.kv_cache_dtype != "auto":
        return "fp8-prefill", "kv_cache_dtype=%r, expected 'auto'" % (layer.kv_cache_dtype,)
    if q_dtype is not None and q_dtype != q.dtype:
        return "fp8-prefill", "prefill q_data_type=%s but q is %s" % (q_dtype, q.dtype)
    if layer.dcp_q_replicate:
        return "dcp", "dcp_q_replicate is set"
    dcp = getattr(layer.impl, "dcp_world_size", None)
    if dcp != 1:
        return "dcp", "impl.dcp_world_size=%r, expected 1" % (dcp,)
    if layer.use_pcp:
        return "pcp", "use_pcp is set"
    if layer.use_sparse:
        return "sparse", "use_sparse is set"
    if layer.sliding_window is not None:
        return "sliding-window", "sliding_window=%r" % (layer.sliding_window,)
    if output.dtype != torch.bfloat16:
        return "output-dtype", "output is %s, expected torch.bfloat16" % (output.dtype,)
    scale = getattr(layer.impl, "scale", None)
    try:
        scale = float(scale)
    except (TypeError, ValueError):
        return "scale", "impl.scale=%r is not a number" % (scale,)
    if not (math.isfinite(scale) and scale > 0):
        return "scale", "impl.scale=%r is not a finite positive number" % (scale,)
    layer_scale = getattr(layer, "scale", None)
    if layer_scale is not None and abs(float(layer_scale) - scale) > 1e-6 * scale:
        return "scale", "layer.scale=%r differs from impl.scale=%r" % (layer_scale, scale)
    heads = int(layer.num_heads)
    p, r = int(layer.qk_nope_head_dim), int(layer.qk_rope_head_dim)
    v_dim, lora = int(layer.v_head_dim), int(layer.kv_lora_rank)
    if tuple(q.shape) != (n, heads, p + r):
        return "shape", "q.shape=%s, expected %s" % (tuple(q.shape), (n, heads, p + r))
    if kv_c_normed.shape[0] < n or kv_c_normed.shape[-1] != lora:
        return "shape", "kv_c_normed.shape=%s, expected [>=%d, %d]" % (tuple(kv_c_normed.shape), n, lora)
    if k_pe.shape[0] < n or k_pe[:n].numel() != n * r:
        return "shape", "k_pe.shape=%s, expected [>=%d, 1, %d] or [>=%d, %d]" % (
            tuple(k_pe.shape), n, r, n, r)
    if output.numel() != n * heads * v_dim:
        return "shape", "output.numel()=%d, expected %d" % (output.numel(), n * heads * v_dim)
    return None


def _mla_adapter(layer, job, session, n, q, kv_c_normed, k_pe, output):
    """Adapter for an ``MLAAttention`` layer: guards, then rebuilt per-head keys and values.

    Returns
    -------
    tuple or None
        ``(view, q3, k3, v3, out2d, shared_from)`` with ``shared_from == -1``.
        None after failing the job with ``MLA guard <name> failed at <layer>: <detail>``.
    """
    import torch

    failed = _mla_guard(layer, q, kv_c_normed, k_pe, output, n)
    if failed is not None:
        session["failed"] = "MLA guard %s failed at %s: %s" % (failed[0], layer.layer_name, failed[1])
        _log(session["failed"])
        _write_done(job, False, error=session["failed"])
        return None
    heads = int(layer.num_heads)
    p, r = int(layer.qk_nope_head_dim), int(layer.qk_rope_head_dim)
    with torch.no_grad():
        k3, v3 = _mla_kv(layer, kv_c_normed, k_pe, n)
    return _MLAView(layer), q.view(n, heads, p + r), k3, v3, output.view(n, heads * int(layer.v_head_dim)), -1


def _capture_tail(layer, job, session, n, view, q3, k3, v3, out2d, shared_from):
    """Rows or mass of the job's positions for one layer, the gate, and finalization.

    Everything past the layer-class adapters is shared: the view stands in for
    the layer in the row, mass and gate helpers.
    """
    import torch

    name = layer.layer_name
    with torch.no_grad():
        rows, sink = _rows_for(view, q3, k3, job["positions"])
        out_p = out2d[
            torch.tensor(job["positions"], device=out2d.device, dtype=torch.long)
        ]
        max_abs, max_rel = _gate(view, rows, v3, out_p)
        entry = {
            "index": _layer_index(name),
            "has_sink": sink is not None,
            "window": int(view.sliding_window or 0),
            "kv_shared_from": shared_from,
            "attention": getattr(view, "attention", "mha"),
            "view": view,
        }
        if job["mode"] == "mass":
            # The gate rows above validate the same q/k/mask/sink arithmetic
            # the mass reduction uses; only the reduced statistics are kept.
            classes = torch.tensor(job["classes"], device=q3.device, dtype=torch.long)
            mass, entropy, msink, pentropy = _mass_for(view, q3, k3, job["query_start"], classes)
            entry["mass"] = mass.cpu().numpy()
            entry["entropy"] = entropy.cpu().numpy()
            entry["prompt_entropy"] = pentropy.cpu().numpy()
            entry["sink"] = msink.cpu().numpy() if msink is not None else None
            del mass, entropy, msink, pentropy
        else:
            entry["rows"] = rows.to(torch.float16).cpu().numpy()
            entry["sink"] = sink.to(torch.float16).cpu().numpy() if sink is not None else None
    session["captured"][name] = entry
    session["gate"][name] = {"max_abs": max_abs, "max_rel": max_rel}
    if max_rel > _gate_tol():
        session["failed"] = "gate failed at %s: max_rel=%.3g > tol=%.3g (max_abs=%.3g)" % (
            name, max_rel, _gate_tol(), max_abs)
        _log(session["failed"])
        _write_done(job, False, error=session["failed"], gate_max_rel=max_rel)
        return
    if len(session["captured"]) == len(session["names"]):
        _finalize(view, session)


def _maybe_capture(classes, layer, query, key, value, output):
    """Capture hook for a plain ``Attention`` layer (called after its kernel ran).

    Parameters
    ----------
    classes : tuple of type
        All captured attention classes of the model (see ``_attention_layers``).
    layer, query, key, value, output
        What ``Attention.forward`` received and returned; ``key``/``value`` are
        None for a KV-sharing layer.
    """
    n = int(query.shape[0])
    ctx = _capture_prelude(classes, layer, n)
    if ctx is None:
        return
    job, session = ctx
    parts = _attention_adapter(layer, job, session, n, query, key, value, output)
    if parts is not None:
        _capture_tail(layer, job, session, n, *parts)


def _maybe_capture_mla(classes, layer, q, kv_c_normed, k_pe, output):
    """Capture hook for an ``MLAAttention`` layer (called after its kernel ran).

    Same protocol as ``_maybe_capture``; the keys and values are rebuilt from
    the latent and the MLA guards run first (on the job's forward only, never on
    a warm-up or other-length forward).
    """
    n = int(q.shape[0])
    ctx = _capture_prelude(classes, layer, n)
    if ctx is None:
        return
    job, session = ctx
    parts = _mla_adapter(layer, job, session, n, q, kv_c_normed, k_pe, output)
    if parts is not None:
        _capture_tail(layer, job, session, n, *parts)


def _finalize(view, session):
    """Write the job's arrays and meta (upload, then done file) once every layer is captured.

    ``view`` is the last captured layer's view (the layer itself for ``Attention``);
    the per-rank head counts and head size in the meta are read from it. Entries
    carry the ``attention`` kind of their layer, summarised in the meta as "mla",
    "mha" or "mixed".
    """
    import numpy as np

    job = session["job"]
    order = sorted(session["captured"].values(), key=lambda e: e["index"])
    mode = job["mode"]
    arrays = {}
    if mode == "mass":
        arrays["mass"] = np.stack([e["mass"] for e in order])         # [L, P, H, C]
        arrays["entropy"] = np.stack([e["entropy"] for e in order])   # [L, P, H]
        arrays["prompt_entropy"] = np.stack([e["prompt_entropy"] for e in order])   # [L, P, H]
        lead = arrays["entropy"].shape[1:]
        arrays["classes"] = np.asarray(job["classes"], dtype=np.uint8)
        arrays["query_start"] = np.asarray(job["query_start"], dtype=np.int32)
        main_shape = list(arrays["mass"].shape)
    else:
        arrays["rows"] = np.stack([e["rows"] for e in order])         # [L, P, H, T]
        lead = arrays["rows"].shape[1:3]
        main_shape = list(arrays["rows"].shape)
    sink = np.stack([e["sink"] if e["sink"] is not None
                     else np.zeros(lead, dtype=np.float16) for e in order])
    layers = np.asarray([e["index"] for e in order], dtype=np.int32)
    kinds = {e["attention"] for e in order}
    kind = kinds.pop() if len(kinds) == 1 else "mixed"
    meta = {
        "job_id": job["job_id"], "seq": job["seq"], "rank": _rank(),
        "tp": len(job.get("put_urls", [])) or 1,
        "n_tokens": job["n_tokens"], "positions": list(job["positions"]),
        "layers": layers.tolist(), "num_heads_local": int(view.num_heads),
        "num_kv_heads_local": int(view.num_kv_heads), "head_size": int(view.head_size),
        "capture_version": CAPTURE_VERSION, "attention": kind,
        "dtype": "float16", "rows_shape": main_shape, "mode": mode,
        "query_start": job.get("query_start"), "n_classes": N_CLASSES if mode == "mass" else None,
        "gate": {nm: session["gate"][nm] for nm in session["gate"]},
        "gate_tol": _gate_tol(),
        "gate_max_rel": max(g["max_rel"] for g in session["gate"].values()),
        "vllm_commit": os.environ.get("VLLM_BUILD_COMMIT"),
        "model": os.environ.get("SMOLBENCH_ATTN_MODEL"),
        "wall_s": time.time() - session["t0"],
    }
    mla_views = [e["view"] for e in order if e["attention"] == "mla"]
    if mla_views:
        # Per-layer head dims are identical across a model's MLA layers.
        meta["qk_head_dim"] = int(mla_views[0].qk_head_dim)
        meta["v_head_dim"] = int(mla_views[0].v_head_dim)
        meta["kv_lora_rank"] = int(mla_views[0].kv_lora_rank)
        meta["head_order"] = ("local head j of rank r is global head r * num_heads_local + j: "
                              "concatenate the ranks' arrays in rank order along the head axis")
    buf = io.BytesIO()
    np.savez(
        buf, sink=sink, layers=layers, **arrays,
        positions=np.asarray(job["positions"], dtype=np.int32),
        has_sink=np.asarray([e["has_sink"] for e in order], dtype=bool),
        sliding_window=np.asarray([e["window"] for e in order], dtype=np.int32),
        kv_shared_from=np.asarray([e["kv_shared_from"] for e in order], dtype=np.int32),
    )
    blob = buf.getvalue()
    meta["bytes"] = len(blob)
    rank = _rank()
    try:
        put_url = job["put_urls"][rank]
        meta_url = job["meta_put_urls"][rank]
    except (KeyError, IndexError, TypeError):
        _write_done(job, False, error="no presigned URLs for rank %d" % rank,
                    gate_max_rel=meta["gate_max_rel"], nbytes=len(blob))
        return
    # The array goes first and the meta only after it landed: a meta json
    # without its npz is a permanently partial job in an append-only bucket
    # (the key exists, so it can never be re-sent), whereas an npz without a
    # meta fails the audit and is reported.
    statuses = [_put(put_url, blob, "application/octet-stream")]
    if _landed(statuses[0]):
        statuses.append(_put(meta_url, json.dumps(meta).encode(), "application/json"))
    ok = len(statuses) == 2 and all(_landed(s) for s in statuses)
    _log("job %s (%s): %d layers, shape %s, %.1f MB, gate max_rel=%.3g, upload=%s"
         % (job["job_id"], mode, len(order), main_shape, len(blob) / 1e6,
            meta["gate_max_rel"], statuses))
    _write_done(job, ok, error=None if ok else "upload failed: %r" % (statuses,),
                gate_max_rel=meta["gate_max_rel"], nbytes=len(blob), uploaded=ok,
                http=statuses)


def _landed(status):
    """True when a PUT status means the object is stored (2xx or landed earlier)."""
    return status == PUT_LANDED_EARLIER or 200 <= status < 300


def _put(url, data, content_type):
    import requests

    last, shaky = 0, False
    for attempt in range(3):
        try:
            resp = requests.put(url, data=data, timeout=900, headers={
                "Content-Type": content_type, "If-None-Match": "*"})
            last = resp.status_code
            if 200 <= last < 300:
                return last
            if last == 412 and attempt:
                if not shaky:
                    return last
                _log("PUT: an earlier attempt landed")
                return PUT_LANDED_EARLIER
            if last == 412:
                return last
            shaky = shaky or last >= 500
        except Exception as exc:  # noqa: BLE001 -- retried
            _log("PUT attempt %d failed: %r" % (attempt, exc))
            last, shaky = -1, True
        time.sleep(2 * (attempt + 1))
    return last

def register():
    """vLLM general-plugin entry point: wrap ``Attention.forward`` (and ``MLAAttention.forward``) once.

    ``MLAAttention`` is imported defensively: a vLLM without the module (or
    without that class at the pinned path) still gets the plain ``Attention``
    capture. Both wrappers share one ``classes`` tuple so every layer of either
    kind is counted before a job finalizes. Idempotent: ``_STATE["patched"]``
    guards both wraps, so a second call leaves both class attributes unchanged.
    """
    if _STATE["patched"]:
        return
    from vllm.model_executor.layers.attention.attention import Attention

    try:
        from vllm.model_executor.layers.attention.mla_attention import MLAAttention
    except ImportError as exc:
        _log("MLAAttention not importable (%r): capturing Attention layers only" % (exc,))
        MLAAttention = None
    classes = (Attention,) if MLAAttention is None else (Attention, MLAAttention)

    original = Attention.forward

    def forward(self, query, key, value, output_shape=None, output_dtype=None):
        output = original(self, query, key, value, output_shape=output_shape,
                          output_dtype=output_dtype)
        try:
            _maybe_capture(classes, self, query, key, value, output)
        except Exception as exc:  # noqa: BLE001 -- serving must never break
            _capture_error(self, exc)
        return output

    Attention.forward = forward
    if MLAAttention is not None:
        mla_original = MLAAttention.forward

        def mla_forward(self, *args, **kwargs):
            # Same arguments through untouched; q, kv_c_normed, k_pe are read by
            # position or keyword so the call convention of vLLM may change freely.
            output = mla_original(self, *args, **kwargs)
            try:
                names = ("q", "kv_c_normed", "k_pe")
                q, kv_c_normed, k_pe = (args[i] if len(args) > i else kwargs[nm]
                                        for i, nm in enumerate(names))
                _maybe_capture_mla(classes, self, q, kv_c_normed, k_pe, output)
            except Exception as exc:  # noqa: BLE001 -- serving must never break
                _capture_error(self, exc)
            return output

        MLAAttention.forward = mla_forward
    _STATE["patched"] = True
    _log("registered (attn dir %s, gate tol %g, classes %s)"
         % (_attn_dir(), _gate_tol(), [c.__name__ for c in classes]))


def _capture_error(layer, exc):
    """Record an unexpected capture failure (log, and fail the job when one is running)."""
    job = _STATE["job"]
    _log("capture error at %s: %r" % (getattr(layer, "layer_name", "?"), exc))
    if job is not None and _STATE["session"] is not None:
        _write_done(job, False, error="%s: %s" % (type(exc).__name__, exc))
